\begin{center}

\begin{longtable}{| m{3cm} | m{0.75\textwidth} |}
\hline
\rowcolor{gray!60} \textbf{Termine} & \textbf{Descrizione} \\ \hline
\rowcolor{white} \textbf{Attore principale} & Attore che interagisce attivamente con il sistema. \\ \hline
\rowcolor{gray!20} \textbf{Pre-condizioni} & Lista di elementi necessari affinché l'utente possa eseguire il caso d'uso. \\ \hline
\rowcolor{white} \textbf{Post-condizioni} & Stato finale del sistema dopo la conclusione del caso d’uso. \\ \hline

\rowcolor{white} \textbf{Inclusioni} & Caso d'uso eseguito in modo incondizionato durante il principale. \\ \hline
\rowcolor{gray!20} \textbf{Estensioni} & Caso d'uso eseguito solo in condizioni particolari. \\ \hline
\end{longtable}
\end{center}

\subsubsection{UC-BW1 - Esecuzione Walkthrough Interattivo} \label{subsec:UC-BW1}

\begin{figure}[H] 
    \centering
    % FIX IMMAGINE: Percorso relativo esplicito
    \includegraphics[width=0.8\textwidth]{../../../resources/UC-BW1.png}
    \caption{Diagramma del caso d'uso UC-BW1}
    \label{fig:uc-bw1}
\end{figure}

\begin{itemize}
    \item \textbf{Attore principale:} Utente Generico
    
    \item \textbf{Pre-condizioni:} 
    \begin{itemize}
        \item L’utente non ha mai aperto l’applicazione
        \item L’utente non ha ancora visualizzato tutte le pagine del walkthrough
    \end{itemize}
    
    \item \textbf{Post-condizioni:}
    \begin{itemize}
        \item L’utente ha visualizzato le funzionalità principali
        \item L’utente ha completato le configurazioni iniziali (obbligatorie o facoltative)
        \item Il sistema mostra la Dashboard principale
    \end{itemize}
    
    \item \textbf{Scenario principale:}
    \begin{itemize}
        \item Il sistema mostra la schermata di benvenuto
        \item L'utente esegue la profilazione iniziale $\rightarrow$ \ref{subsec:UC-BW1.2}
        \item Il sistema inizia il ciclo di presentazione delle schede informative dell'app
        \item L'utente interagisce con la scheda $\rightarrow$ \ref{subsec:UC-BW1.5}, \ref{subsec:UC-BW1.6}, \ref{subsec:UC-BW1.7}
        \item L'utente naviga verso la scheda successiva appena ha finito di interagire/leggere la scheda $\rightarrow$ \ref{subsec:UC-BW1.3}
        \item Il sistema chiude il walkthrough e apre la Home Page
    \end{itemize}
    
    \item \textbf{Scenario secondario:}
    \begin{itemize}
        \item L'utente decide di tornare alla scheda precedente $\rightarrow$ \ref{subsec:UC-BW1.4}
        \item L'utente decide di saltare il tutorial $\rightarrow$ \ref{subsec:UC-BW1.1}
    \end{itemize}
    
    \item \textbf{Inclusioni:}
    \begin{itemize}
        \item \ref{subsec:UC-BW1.2}
        \item \ref{subsec:UC-BW1.3}
    \end{itemize}
    
    \item \textbf{Estensioni:}
    \begin{itemize}
        \item \ref{subsec:UC-BW1.1}
        \item \ref{subsec:UC-BW1.4}
        \item \ref{subsec:UC-BW1.5}
        \item \ref{subsec:UC-BW1.6}
        \item \ref{subsec:UC-BW1.7}
    \end{itemize}
    
    \item \textbf{Trigger:} Primo avvio dell'applicazione
\end{itemize}

\subsubsection{UC-BW1.1 - Salta Walkthrough} \label{subsec:UC-BW1.1}
\begin{itemize}
    \item \textbf{Attore principale:} Utente Generico
    
    \item \textbf{Pre-condizioni:}
    \begin{itemize}
        \item Il walkthrough è in corso
    \end{itemize}
    
    \item \textbf{Post-condizioni:}
    \begin{itemize}
        \item Accesso diretto alla Dashboard senza ulteriori configurazioni
    \end{itemize}
    
    \item \textbf{Scenario principale:}
    \begin{itemize}
        \item L'utente seleziona la voce "Salta" durante il walkthrough
        \item Il sistema interrompe la presentazione delle schede e apre direttamente la Home Page
    \end{itemize}
    
    \item \textbf{Trigger:} L'utente preme il pulsante "Salta"
\end{itemize}

\begin{itemize}
    \item \textbf{Attore principale:} Utente Generico
    
    \item \textbf{Pre-condizioni:}
    \begin{itemize}
        \item Il walkthrough è attivo e sta presentando la scheda di benvenuto
    \end{itemize}
    
    \item \textbf{Post-condizioni:}
    \begin{itemize}
        \item Il contesto d'uso è salvato e il sistema adatta i testi e i contenuti successivi in base alla scelta
    \end{itemize}
    
    \item \textbf{Scenario principale:}
    \begin{itemize}
        \item L'utente seleziona l'opzione "Per me" o "Per altri"
    \end{itemize}
    
    \item \textbf{Trigger:} Il sistema richiede la selezione del profilo mostrando la relativa schermata
\end{itemize}

    \item \textbf{Post-condizioni:}

% --- UC-BW1.5 ---
\subsubsection{UC-BW1.5 - Configurazione Rapida Contatto} \label{subsec:UC-BW1.5}
\begin{itemize}
    \item \textbf{Attore principale:} Utente Generico
    
    \item \textbf{Pre-condizioni:}
    \begin{itemize}
        \item Il walkthrough è attivo
        \item La scheda riguardante la "Rete di Contatti Fidati" è visibile
    \end{itemize}
    
    \item \textbf{Post-condizioni:}
    \begin{itemize}
        \item Il contatto è salvato in memoria OPPURE nessun contatto è salvato e l'utente è informato
    \end{itemize}
    
    \item \textbf{Scenario principale:}
    \begin{itemize}
        \item L'utente compila i campi presenti nella scheda
        \item Il sistema valida e salva il contatto immediatamente
    \end{itemize}
    
    \item \textbf{Scenario secondario:}
    \begin{itemize}
        \item L'utente lascia i campi vuoti e non interagisce
        \item L'utente preme "Avanti" $\rightarrow$ \ref{subsec:UC-BW1}
        \item Il sistema non salva nulla
        \item Il sistema mostra un breve avviso "Potrai configurare i contatti più tardi nelle Impostazioni"
    \end{itemize}
    
    \item \textbf{Trigger:} L'utente interagisce con i campi di input della scheda Contatti
\end{itemize}

% --- UC-BW1.6 ---
\subsubsection{UC-BW1.6 - Configurazione Rapida Email} \label{subsec:UC-BW1.6}
\begin{itemize}
    \item \textbf{Attore principale:} Utente Generico
    
    \item \textbf{Pre-condizioni:}
    \begin{itemize}
        \item Il walkthrough è attivo
        \item La scheda riguardante la funzionalità di dead man’s switch è visibile
    \end{itemize}
    
    \item \textbf{Post-condizioni:}
    \begin{itemize}
        \item L'email è salvata OPPURE nessuna email è salvata e l'utente è informato
    \end{itemize}
    
    \item \textbf{Scenario principale:}
    \begin{itemize}
        \item L'utente inserisce la propria email
        \item Il sistema valida il formato
        \item Il sistema salva l'indirizzo
    \end{itemize}
    
    \item \textbf{Scenario secondario:}
    \begin{itemize}
        \item L'utente lascia il campo vuoto
        \item Il sistema non salva nulla
        \item Il sistema notifica l’utente del fatto che non potrà usare la funzionalità ma che può configurarla nelle impostazioni
    \end{itemize}
    
    \item \textbf{Trigger:} L'utente interagisce con il campo di input della scheda Email
\end{itemize}

    \end{itemize}

    \end{itemize}
    
    \item \textbf{Trigger:} L'utente interagisce con i campi di input della scheda Diario
\end{itemize}
